cmake_minimum_required(VERSION 3.20)
project(ChartDifficultyCalculator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# MinGW's assembler can corrupt non-ASCII object subdirectories produced for
# out-of-tree source paths. Configure-time copies keep object paths ASCII while
# CMake tracks changes to the original shared implementation for regeneration.
set(shared_sources)
foreach(source pattern_difficulty.cpp mania_strain.cpp dan_rating.cpp)
  configure_file("../src/${source}" "shared/${source}" COPYONLY)
  list(APPEND shared_sources "${CMAKE_CURRENT_BINARY_DIR}/shared/${source}")
endforeach()
add_library(calculator_core STATIC src/chart_reader.cpp src/analysis.cpp ${shared_sources})
target_include_directories(calculator_core PUBLIC src ../src)
target_compile_definitions(calculator_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(ChartDifficultyCalculator WIN32 src/main.cpp)
target_link_libraries(ChartDifficultyCalculator PRIVATE calculator_core user32 gdi32 comdlg32 shell32)
if(MINGW)
  target_compile_options(calculator_core PRIVATE -Wall -Wextra)
  target_compile_options(ChartDifficultyCalculator PRIVATE -Wall -Wextra)
  target_link_options(ChartDifficultyCalculator PRIVATE -static -static-libgcc -static-libstdc++ -municode)
endif()
if(MSVC)
  target_compile_options(calculator_core PRIVATE /utf-8)
  target_compile_options(ChartDifficultyCalculator PRIVATE /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(chart_reader_test tests/chart_reader_test.cpp)
  target_link_libraries(chart_reader_test PRIVATE calculator_core shell32)
  add_executable(analysis_test tests/analysis_test.cpp)
  target_link_libraries(analysis_test PRIVATE calculator_core)
  if(MINGW)
    target_link_options(chart_reader_test PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(analysis_test PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME chart_reader COMMAND chart_reader_test)
  add_test(NAME chart_analysis COMMAND analysis_test)
  add_test(NAME calculator_ui COMMAND ChartDifficultyCalculator --self-test)
  set_tests_properties(chart_reader chart_analysis calculator_ui PROPERTIES TIMEOUT 30)
endif()
